# rownames 로 쓰기 위해 data.frame 으로 변환

beer_df <- as.data.frame(beer)

rownames(beer_df) <- beer_df$pname

# 분석에 사용할 변수만 추출

X <- beer_df[, c("cal", "sod", "alc", "cost")]

# NA 제거(있으면)

X <- na.omit(X)

# 표준화 (평균 0, 분산 1)

Z <- scale(X)
Z <- as.data.frame(Z)

# 유클리드 거리

DZ <- dist(Z)

# Ward.D2 방법

hc_ward <- hclust(DZ, method = "ward.D2")

plot(hc_ward, hang = -1, main = "Dendrogram (Ward.D2)")
rect.hclust(hc_ward, k = 3, border = 2:4)   #여기서는 k=3 예시

# 3개 군집으로 cut

k_hc <- cutree(hc_ward, k = 3)
table(k_hc)
## k_hc

## 9 4 7
# 실루엣 계수
